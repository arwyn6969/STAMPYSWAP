(() => {
  const $$ = (selector, root = document) => [...root.querySelectorAll(selector)];
  const projects = $$('[data-project]').map(element => {
    const link = element.matches('a') ? element : element.querySelector('[data-launch]');
    return { element, name: element.dataset.project, category: element.dataset.category, search: element.dataset.search || '', description: element.querySelector('[data-description]').textContent.trim(), href: link?.getAttribute('href'), external: link?.target === '_blank', random: element.hasAttribute('data-random'), warning: element.classList.contains('danger'), status: element.dataset.status };
  });
  const categoryLabels = {art:'Art & collections', play:'Play & wander', tools:'Onchain tools', lab:'In the lab'};
  const searchDialog = document.getElementById('search-dialog');
  const searchInput = document.getElementById('project-search');
  const searchResults = document.getElementById('search-results');
  let previousFocus = null;
  function openDialog(dialog, trigger) {
    previousFocus = trigger || document.activeElement;
    if (dialog === searchDialog) { searchInput.value = ''; renderSearch(); }
    dialog.showModal();
    if (dialog === searchDialog) searchInput.focus();
  }
  $$('[data-open]').forEach(button => button.addEventListener('click', () => openDialog(document.getElementById(button.dataset.open), button)));
  $$('dialog').forEach(dialog => {
    dialog.addEventListener('keydown', event => {
      if (event.key === 'Escape') { event.preventDefault(); dialog.close(); }
    });
    dialog.querySelector('[data-close]').addEventListener('click', () => dialog.close());
    dialog.addEventListener('click', event => {
      const box = dialog.getBoundingClientRect();
      if (event.target === dialog && (event.clientX < box.left || event.clientX > box.right || event.clientY < box.top || event.clientY > box.bottom)) dialog.close();
    });
    dialog.addEventListener('close', () => { if (previousFocus?.isConnected) previousFocus.focus({preventScroll:true}); });
  });
  function filterProjects(category) {
    $$('[data-filter]').forEach(button => button.setAttribute('aria-pressed', String(button.dataset.filter === category)));
    document.querySelector('.folder-grid').classList.toggle('is-filtered', category !== 'all');
    $$('[data-folder]').forEach(folder => { folder.hidden = category !== 'all' && folder.dataset.folder !== category; });
    const count = projects.filter(project => category === 'all' || project.category === category).length;
    document.getElementById('filter-status').textContent = `${count} projects${category === 'all' ? ' across all folders' : ' in ' + categoryLabels[category]}.${category === 'art' || category === 'all' ? ' Rare Whales is pinned above.' : ''}`;
  }
  $$('[data-filter]').forEach(button => button.addEventListener('click', () => filterProjects(button.dataset.filter)));
  // Index the real links already in the HTML; browsing also works without JavaScript.
  function renderSearch() {
    const words = searchInput.value.trim().toLowerCase().split(/\s+/).filter(Boolean);
    const matches = projects.filter(project => words.every(word => `${project.name} ${project.description} ${project.category} ${project.search}`.toLowerCase().includes(word)));
    searchResults.replaceChildren();
    for (const project of matches) {
      const row = document.createElement(project.href ? 'a' : 'div');
      row.className = 'search-result';
      if (project.href) {
        row.href = project.href;
        if (project.external) { row.target = '_blank'; row.rel = 'noopener noreferrer'; }
      }
      const name = document.createElement('span');
      name.textContent = project.name;
      const meta = document.createElement('small');
      meta.textContent = !project.href ? 'Coming soon' : project.warning ? '⚠ Unstable · Read warning' : project.status ? `${project.status} →` : `${categoryLabels[project.category]} ${project.external ? '↗' : '→'}`;
      if (project.external) row.setAttribute('aria-label', `${project.name}, ${categoryLabels[project.category]} (opens in a new tab)`);
      row.append(name, meta);
      searchResults.append(row);
    }
    if (!matches.length) { const empty = document.createElement('p'); empty.className = 'empty-state'; empty.textContent = 'No files found. Try art, Bitcoin, or paint.'; searchResults.append(empty); }
    document.getElementById('search-status').textContent = `${matches.length} ${matches.length === 1 ? 'project' : 'projects'} found.`;
  }
  searchInput.addEventListener('input', renderSearch);
  searchDialog.addEventListener('keydown', event => {
    const links = $$('a.search-result', searchResults);
    const index = links.indexOf(document.activeElement);
    if ((event.key === 'ArrowDown' || event.key === 'ArrowUp') && (document.activeElement === searchInput || index !== -1) && links.length) {
      event.preventDefault();
      const next = event.key === 'ArrowDown' ? (index + 1) % links.length : index <= 0 ? links.length - 1 : index - 1;
      links[next].focus();
    }
    if (event.key === 'Enter' && document.activeElement === searchInput && links.length) { event.preventDefault(); links[0].click(); }
  });
  document.addEventListener('keydown', event => {
    const typing = event.target instanceof Element && (event.target.closest('input, textarea, select') || event.target.isContentEditable);
    if ((event.metaKey || event.ctrlKey) && event.key.toLowerCase() === 'k' || event.key === '/' && !typing) {
      if (document.querySelector('dialog[open]')) return;
      event.preventDefault(); openDialog(searchDialog);
    }
  });
  if (!/Mac|iPhone|iPad/.test(navigator.platform)) $$('[data-shortcut]').forEach(kbd => { kbd.textContent = 'Ctrl K'; });
  const randomProjects = projects.filter(project => project.random && project.href && !project.warning);
  let lastPick;
  function pickRandom() {
    const pool = randomProjects.filter(project => project !== lastPick);
    const project = pool[Math.floor(Math.random() * pool.length)];
    lastPick = project;
    document.getElementById('random-title').textContent = project.name;
    document.getElementById('random-description').textContent = project.description;
    const launch = document.getElementById('random-launch');
    launch.href = project.href;
    launch.target = project.external ? '_blank' : '_self';
    launch.rel = 'noopener noreferrer';
    document.getElementById('random-external').hidden = !project.external;
  }
  $$('[data-random-open]').forEach(button => button.addEventListener('click', () => { pickRandom(); openDialog(document.getElementById('random-dialog'), button); }));
  document.getElementById('random-again').addEventListener('click', pickRandom);
  const clock = document.getElementById('clock');
  const timeFormat = new Intl.DateTimeFormat('en-GB', {timeZone:'Europe/London', hour:'2-digit', minute:'2-digit', second:'2-digit'});
  function updateClock() { clock.textContent = timeFormat.format(new Date()); }
  updateClock(); setInterval(updateClock, 1000);
})();
</script>
